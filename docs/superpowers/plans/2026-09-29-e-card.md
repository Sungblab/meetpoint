# 전자 명함 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 모든 참가자가 전화·이메일·기타 연락처로 된 전자 명함을 갖고, 서로 연결되면 명찰 뒷면에서 명함을 보고 한 번에 휴대폰 연락처(vCard)로 저장하게 한다.

**Architecture:** `participants`에 `phone`, `email` 칸을 더하고 시작할 때 없는 칸만 붙인다. 연락처 세 칸은 기존 `contact`와 같은 규칙(서로 연결됐을 때만)으로 API에 싣고, vCard는 새 엔드포인트가 만든다. 화면은 명찰 뒷면의 전자 명함과 입장 양식의 미리보기, 명함 사진 읽기 확장으로 이뤄진다.

**Tech Stack:** FastAPI, SQLAlchemy async(SQLite 개발·Postgres 운영), Jinja 템플릿, 순수 JS, pytest.

**Spec:** `docs/superpowers/specs/2026-09-29-e-card-design.md`

## Global Constraints

- 새 의존성을 추가하지 않는다.
- 사용자 입력은 JS에서 `textContent`/`setAttribute`로만 넣는다. `innerHTML` 금지 (CLAUDE.md).
- 전화·이메일·기타 연락처와 vCard는 서로 ‘만나고 싶어요’를 누른 경우에만 내보낸다 (CLAUDE.md의 연락처 규칙).
- 연락처 정보는 Gemini 추천 입력에 넣지 않는다. `Profile.as_prompt_dict`는 바꾸지 않는다.
- 색은 관계와 연결 상태에만 쓴다 (DESIGN.md). 전자 명함의 강조는 연결 상태색(`--match`)과 참가자 회색(`--attendee`)만.
- 화면 문구는 해요체.
- 테스트는 `uv run pytest`. 각 작업이 끝날 때 전체가 통과해야 한다.

    async def create_all(self) -> None:
        from . import models  # noqa: F401  테이블 등록

        async with self.engine.begin() as conn:
            await conn.run_sync(Base.metadata.create_all)
            await conn.run_sync(_add_missing_columns)


def _add_missing_columns(sync_conn) -> None:
    """create_all은 이미 있는 테이블에 칸을 더하지 않는다. 모델에 새로 생긴 문자열 칸을 기본값 ''로 붙인다."""
    # ponytail: 문자열 칸만 자동으로 붙인다. 다른 타입이 필요해지면 Alembic을 들인다.
    insp = inspect(sync_conn)
    for table in Base.metadata.sorted_tables:
        if not insp.has_table(table.name):
            continue
        existing = {c["name"] for c in insp.get_columns(table.name)}
        for col in table.columns:
            if col.name in existing:
                continue
            if not isinstance(col.type, String):
                raise RuntimeError(f"{table.name}.{col.name}: 문자열이 아닌 새 칸은 직접 마이그레이션해야 해요")
            ddl = col.type.compile(dialect=sync_conn.dialect)
            sync_conn.execute(text(f"ALTER TABLE {table.name} ADD COLUMN {col.name} {ddl} NOT NULL DEFAULT ''"))
```

"""서로 연결된 사람의 전자 명함을 휴대폰 연락처 파일(vCard 3.0)로 만든다."""

from urllib.parse import quote


def _esc(value: str) -> str:
    # RFC 6350 3.4: 역슬래시, 쉼표, 세미콜론, 줄바꿈을 이스케이프한다.
    return (
        value.replace("\\", "\\\\")
        .replace(",", "\\,")
        .replace(";", "\\;")
        .replace("\r\n", "\\n")
        .replace("\n", "\\n")
    )


def build_vcard(*, name: str, headline: str, phone: str, email: str, other: str, note: str) -> str:
    lines = ["BEGIN:VCARD", "VERSION:3.0", f"N:;{_esc(name)};;;", f"FN:{_esc(name)}"]
    if headline:
        lines.append(f"TITLE:{_esc(headline)}")
    if phone:
        lines.append(f"TEL;TYPE=CELL:{_esc(phone)}")
    if email:
        lines.append(f"EMAIL;TYPE=INTERNET:{_esc(email)}")
    memo = note + (f"\n기타 연락처: {other}" if other else "")
    lines.append(f"NOTE:{_esc(memo)}")
    lines.append("END:VCARD")
    return "\r\n".join(lines) + "\r\n"


def vcard_disposition(name: str) -> str:
    # 한글 이름은 filename*로, 오래된 브라우저에는 ASCII 이름을 준다.
    return f"attachment; filename=\"contact.vcf\"; filename*=UTF-8''{quote(name + '.vcf')}"
```

    mine = await s.scalar(select(Like.id).where(Like.from_id == a, Like.to_id == b))
    theirs = await s.scalar(select(Like.id).where(Like.from_id == b, Like.to_id == a))

@router.get("/r/{code}/api/vcard/{target_id}")
async def vcard(request: Request, code: str, target_id: int):
    svc = services(request)
    async with svc.db.sessionmaker() as s:
        room, me = await _auth(request, s, code)
        if room is None or me is None:
            return JSONResponse({"error": "unauthorized"}, status_code=401)
        target = await s.get(Participant, target_id)
        if target is None or target.room_id != room.id or not await _is_matched(s, me.id, target.id):
            return JSONResponse({"error": "not_found"}, status_code=404)
        rec = await s.scalar(
            select(Recommendation).where(Recommendation.viewer_id == me.id, Recommendation.target_id == target.id)
        )
        created = room.created_at if room.created_at.tzinfo else room.created_at.replace(tzinfo=UTC)
        note = f"Meetpoint · {room.name} · {created.astimezone(KST):%Y-%m-%d}"
        if rec is not None and rec.reason:
            note += f" · 만난 이유: {rec.reason}"
        body = build_vcard(name=target.name, headline=target.headline, phone=target.phone,
                           email=target.email, other=target.contact, note=note)
        name = target.name
    return Response(body, media_type="text/vcard; charset=utf-8",
                    headers={"Content-Disposition": vcard_disposition(name), "Cache-Control": "no-store"})

// 입장 양식을 적는 동안 위쪽 명찰과 아래 전자 명함 미리보기를 채운다. textContent만 쓴다.

  const value = (name) => (form.elements[name]?.value || "").trim();

  function sync() {
    for (const node of document.querySelectorAll("[data-preview]")) {

      node.textContent = v || node.dataset.blank || "";

})();
